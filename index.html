// 1. የላክኸው HTML ውስጥ የሚገኘውን #kpi-batch-container በ KPIs እና በ Plan/Actual Input መስኮቶች የሚሞላ function function renderBatchKpiForm() { const container = document.getElementById('kpi-batch-container'); if (!container) return; if (!window.kpiCategories || window.kpiCategories.length === 0) { container.innerHTML = `

ምንም አይነት KPI አልተገኘም!

`; return; } container.innerHTML = kpiCategories.map((kpi, index) => `
${index + 1}. ${kpi}
`).join(''); } // 2. የላክኸው form id="daily-entry-form" onsubmit="submitBatchEntries(event)" ሲደረግ ዳታውን የሚቀበል function function submitBatchEntries(e) { e.preventDefault(); const dateVal = document.getElementById('entry-date').value; const periodVal = document.getElementById('entry-period').value; const staffVal = document.getElementById('entry-staff-select').value; const supervisorVal = document.getElementById('entry-supervisor-select').value; const remarksVal = document.getElementById('entry-remarks').value.trim(); const isAutoApproved = state.currentUser && (state.currentUser.role === 'Super Admin' || state.currentUser.role === 'Admin'); let savedCount = 0; // የቆዩትን ዳታዎች ሳያጠፋ አዳዲሶቹን ብቻ ይጨምራል kpiCategories.forEach((kpi, index) => { const planInput = document.getElementById(`plan_${index}`); const actualInput = document.getElementById(`actual_${index}`); const planVal = planInput ? (parseFloat(planInput.value) || 0) : 0; const actualVal = actualInput ? (parseFloat(actualInput.value) || 0) : 0; // Plan ወይም Actual > 0 ከሆነ ብቻ ይመዘግበዋል if (planVal > 0 || actualVal > 0) { const entryData = { id: 'entry_' + Date.now() + '_' + index, date: dateVal, period: periodVal, staff: staffVal, kpi: kpi, plan: planVal, actual: actualVal, supervisor: supervisorVal, remarks: remarksVal, status: isAutoApproved ? 'Approved' : 'Pending', timestamp: Date.now() }; // የቀደመውን state ሳይነካ አዲሱን መረጃ ከፊት ይጨምራል state.entries.unshift(entryData); // Firebase / Realtime DB ካለ እዚያ ላይ ይመዘግባል if (typeof db !== 'undefined' && db) { db.ref('entries/' + entryData.id).set(entryData); } savedCount++; } }); if (savedCount === 0) { if (typeof showToast === 'function') { showToast("እባክዎ ቢያንስ ለአንዱ KPI ፕላን ወይም አቺቭመንት ያስገቡ!", false); } else { alert("እባክዎ ቢያንስ ለአንዱ KPI ፕላን ወይም አቺቭመንት ያስገቡ!"); } return; } if (typeof syncLocalStorage === 'function') syncLocalStorage(); if (typeof showToast === 'function') { showToast(`${savedCount} የ KPI መረጃዎች በስኬት ተመዝግበዋል!`); } else { alert(`${savedCount} የ KPI መረጃዎች በስኬት ተመዝግበዋል!`); } resetEntryForm(); if (typeof switchTab === 'function') switchTab('dashboard', 'Dashboard Overview'); if (typeof renderDashboard === 'function') renderDashboard(); } // 3. Form ሲላክ ወይም Reset ሲደረግ ፎርሙን የሚያፀዳ function function resetEntryForm() { const form = document.getElementById('daily-entry-form'); if (form) form.reset(); const dateInput = document.getElementById('entry-date'); if (dateInput) dateInput.valueAsDate = new Date(); const staffSelect = document.getElementById('entry-staff-select'); if (staffSelect && state.currentUser) staffSelect.value = state.currentUser.name; // የገባውን Input ዳታ አጽድቶ እንደገና ይስላል renderBatchKpiForm(); }
